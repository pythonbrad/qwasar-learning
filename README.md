### Notice

I autogenerated this repo through the following code.

```bash
#!/usr/bin/bash
#
# Combine my QWASAR work repo in monorepo while keeping the git histories.
#
# I have been able to achieve it through this article
# https://www.delftstack.com/howto/git/git-merge-repositories/
#

set -e

output_folder=combine
readme_file=$(mktemp)

# Create the combined repo
rm -rf $output_folder
mkdir $output_folder
cd $output_folder

# Initialize the repo
git init
git remote add origin git@github.com:pythonbrad/qwasar-learning.git
git branch -m main

# Initialize the README
cat >> $readme_file <<EOF
# My Qwasar Learning

This repo contains my solution to the qwasar projects.

### Tasks

EOF

# Merge the qwasar work repos
for repo in ../*/; do
	echo Merging of $repo...

	origin=$(basename $repo)
	# Make sure to ignore the output folder
	if [ "$origin" == "$output_folder" ]; then continue; fi

	url=`(cd $repo && git remote get-url origin)`
	branch=$origin/dev
	
	git remote add $origin $repo;
	git fetch $origin
	git merge $branch --allow-unrelated-histories --strategy=recursive --no-edit

	mkdir $origin
	git ls-tree --full-tree --name-only $branch | xargs -i git mv -v {} $origin
	git commit -m "Moved $origin to subfolder"

	project_name=`echo $origin | sed -e 's/[-_]/ /g' -e 's/[^ ]\+/\L\u&/g'`
	echo "- [ ] [$project_name]($origin)" >> $readme_file
	echo "" >> $readme_file
	echo -n "    **TASK**: " >> $readme_file
	grep -A 2 Task $repo/README.md | tail -n +2 | sed 's/#.*//g' >> $readme_file
	echo "" >> $readme_file
done

# Complete the readme with notice and license.
cat >> $readme_file <<EOF

### Notice

I autogenerated this repo through the following code.

\`\`\`bash
`cat ../$0`
\`\`\`

### LICENSE

The source code is under the [MIT No Attribution](LICENSE) license.
EOF

# Finalize the README
mv $readme_file README.md

# Add the license
curl -s https://raw.githubusercontent.com/aws/mit-0/refs/heads/master/MIT-0 > LICENSE

# Submission
git add README.md LICENSE
git commit -m "Add README and license"
git push origin main --force
```
